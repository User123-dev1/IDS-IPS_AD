# ADD NEW METHOD TO HANDLE ML ANOMALIES
# ============================================================================

def _handle_ml_anomaly(self, packet_data, ml_result):
    '''Handle detected ML anomaly'''
    
    # Store anomaly
    anomaly_info = {
        'timestamp': packet_data['timestamp'],
        'src_ip': packet_data['src_ip'],
        'dst_ip': packet_data['dst_ip'],
        'dst_port': packet_data['dst_port'],
        'score': ml_result['anomaly_score'],
        'confidence': ml_result['confidence'],
        'details': ml_result['details']
    }
    self.ml_anomalies.append(anomaly_info)
    
    # Keep only last 100 anomalies
    if len(self.ml_anomalies) > 100:
        self.ml_anomalies = self.ml_anomalies[-100:]
    
    # Log high-confidence anomalies
    if ml_result['confidence'] in ['high', 'medium']:
        print(f"\\n⚠ ML ANOMALY DETECTED (confidence: {ml_result['confidence']})")
        print(f"   {packet_data['src_ip']}:{packet_data['src_port']} → "
              f"{packet_data['dst_ip']}:{packet_data['dst_port']}")
        print(f"   Score: {ml_result['anomaly_score']:.3f}")
        print(f"   {ml_result['details']}")
    
    # Trigger your existing alert system (if you have one)
    if hasattr(self, 'trigger_alert'):
        self.trigger_alert({
            'type': 'ml_anomaly',
            'severity': 'high' if ml_result['confidence'] == 'high' else 'medium',
            'message': ml_result['details'],
            'source': packet_data['src_ip'],
            'destination': packet_data['dst_ip']
        })

# ADD METHOD TO GET ML STATISTICS

def get_ml_statistics(self):
    '''Get ML detection statistics'''
    if not self.ml_enabled or not self.ml_detector:
        return None
    
    detector_stats = self.ml_detector.get_statistics()
    recent_anomalies = self.ml_detector.get_recent_anomalies(limit=10)
    
    return {
        'enabled': self.ml_enabled,
        'total_packets': detector_stats['total_packets'],
        'anomaly_count': detector_stats['anomaly_count'],
        'anomaly_rate': detector_stats['anomaly_rate'],
        'recent_anomalies': recent_anomalies
    }
